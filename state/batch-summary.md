**THE SPLITTER IS THE PUBLISHED ONE AND IT STILL STANDS. The cross-check at the foot of the code block is part of the instrument, not a courtesy**: a splitter reporting a median of 19 and a splitter reporting a median of 26 cannot both be measuring this band, and the one that is wrong is the one whose character class has been re-typed by hand. **A straight apostrophe is 0 in this band, which is how you know.**

```python
import re
def w(t): return len(re.findall(r"[A-Za-z0-9£$’'-]+", t))
def sents(t): return [s for s in re.split(r'(?<=[.!?])["*\]’”)]*\s+', t) if s.strip()]
def paras(t): return [l for l in t.split('\n') if l.strip() and not l.strip().startswith('#')]
def rep(L): L=sorted(L); return f"{len(L)} sentences, median {L[len(L)//2]}, over-60 {round(100*sum(1 for l in L if l>60)/len(L),2)}%"
for n in range(471, 481):
    t = open(f'chapters/volume-10/chapter-{n:04d}.md').read()
    L = [w(s) for s in sents(t)]
    print(n, w(t), rep(L), 'maxpara', max(w(l) for l in paras(t)))
X = [w(s) for n in range(471,481) for s in sents(open(f'chapters/volume-10/chapter-{n:04d}.md').read())]
print(rep(X))
Y = [w(s) for n in range(471,481) for s in re.split(r'(?<=[.!?])[^\s]*\s+', open(f'chapters/volume-10/chapter-{n:04d}.md').read()) if s.strip()]
print('CROSS-CHECK', rep(Y))   # n=1296 median 19 over-60 2.78 — agrees to two sentences
```
